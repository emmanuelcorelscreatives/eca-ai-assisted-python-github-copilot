from pathlib import Path
import sqlite3

from .models import Expense


_SCHEMA = """
CREATE TABLE IF NOT EXISTS expenses (
    id INTEGER PRIMARY KEY CHECK (id > 0),
    amount_minor INTEGER NOT NULL CHECK (amount_minor > 0),
    category TEXT NOT NULL,
    spent_on TEXT NOT NULL,
    note TEXT NOT NULL
)
"""


class SQLiteExpenseRepository:
    def __init__(self, db_path: Path) -> None:
        self._db_path = db_path

    def initialize(self) -> None:
        self._db_path.parent.mkdir(parents=True, exist_ok=True)
        connection = sqlite3.connect(self._db_path)
        try:
            connection.execute("BEGIN")
            connection.execute(_SCHEMA)
            connection.commit()
        except Exception:
            connection.rollback()
            raise
        finally:
            connection.close()

    def add(self, expense: Expense) -> int:
        connection = sqlite3.connect(self._db_path)
        try:
            connection.execute("BEGIN")
            cursor = connection.execute(
                """
                INSERT INTO expenses
                    (amount_minor, category, spent_on, note)
                VALUES (?, ?, ?, ?)
                """,
                (
                    expense.amount_minor,
                    expense.category,
                    expense.spent_on,
                    expense.note,
                ),
            )
            expense_id = cursor.lastrowid
            if not isinstance(expense_id, int) or expense_id <= 0:
                raise RuntimeError(
                    "SQLite did not return a positive expense identifier."
                )
            connection.commit()
            return expense_id
        except Exception:
            connection.rollback()
            raise
        finally:
            connection.close()

from pathlib import Path
import sqlite3

from expense_tracker.models import Expense
from expense_tracker.repository import SQLiteExpenseRepository


db_path = Path("data/demo.db")
db_path.unlink(missing_ok=True)

repository = SQLiteExpenseRepository(db_path)
repository.initialize()

expense = Expense(
    amount_minor=1299,
    category="Meals",
    spent_on="2025-03-08",
    note="Client's lunch",
)

first_id = repository.add(expense)
second_id = repository.add(expense)

connection = sqlite3.connect(db_path)
try:
    row = connection.execute(
        "SELECT COUNT(*) FROM expenses"
    ).fetchone()
    row_count = row[0]
finally:
    connection.close()

print(f"saved identifiers: {first_id}, {second_id}")
print(f"stored rows: {row_count}")
