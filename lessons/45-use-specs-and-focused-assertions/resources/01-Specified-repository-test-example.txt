expense_service.py

from dataclasses import dataclass
from decimal import Decimal


@dataclass(frozen=True)
class Expense:
    amount: Decimal
    category: str


class ExpenseRepository:
    def save(self, expense: Expense) -> int:
        raise NotImplementedError


def add_expense(repository: ExpenseRepository, expense: Expense) -> int:
    if expense.amount <= 0:
        raise ValueError("amount must be positive")
    return repository.save(expense)


tests/test_expense_service.py

from decimal import Decimal
from unittest.mock import create_autospec

import pytest

from expense_service import Expense, ExpenseRepository, add_expense


def test_valid_expense_is_saved_once():
    repository = create_autospec(ExpenseRepository, instance=True, spec_set=True)
    expense = Expense(Decimal("12.50"), "travel")
    repository.save.return_value = 7

    result = add_expense(repository, expense)

    assert result == 7
    repository.save.assert_called_once_with(expense)


def test_invalid_expense_is_not_saved():
    repository = create_autospec(ExpenseRepository, instance=True, spec_set=True)
    expense = Expense(Decimal("0"), "travel")

    with pytest.raises(ValueError, match="amount must be positive"):
        add_expense(repository, expense)

    repository.save.assert_not_called()
